# LOCAL DEV

# Backend — local fakes mode (UseLocalFakes: true in appsettings.Local.json,
# no Azure services, FakeTraitsExtractor / InMemoryGarmentRepository / etc.)
ASPNETCORE_ENVIRONMENT=Local dotnet run --project robe.api

# Backend — dev Azure stage config (real Key Vault / OpenAI / Blob, requires
# the dev stage to be deployed and az login with the right subscription)
ASPNETCORE_ENVIRONMENT=Dev dotnet run --project robe.api

# Frontend — local dev server (reads vestra/apps/web/.env.local)
# Copy .env.local.example to .env.local first (leave NEXT_PUBLIC_ENTRA_CLIENT_ID
# commented out to use FakeAuthProvider + local API, no Entra tenant needed)
cd vestra && pnpm dev
# or from repo root:
pnpm --filter @vestra/web run dev

###############################################################################
# DEV STAGE — full end-to-end deploy (infra + backend + frontend)
###############################################################################

cd infra/Azure/bicep

# Deploy everything: resource group, Key Vault, App Service, Azure OpenAI,
# Application Insights, Static Web App, RBAC grants, secrets, robe.api zip
# deploy, smoke test, Next.js build, SWA deploy.
./dev-create.sh

# With flags:
./dev-create.sh --yes                          # skip all confirmation prompts
./dev-create.sh --yes --subscription=<id>      # pick a specific subscription
./dev-create.sh --yes --skip-frontend          # deploy infra+backend only, no frontend build

# Teardown dev (stops cost accrual). Always use --purge-vault --purge-openai
# so the soft-deleted names (kv-robeai-dev, aoai-robe-dev) are freed immediately
# and dev-create.sh can redeploy without a name conflict.
./dev-teardown.sh --yes --wait --purge-vault --purge-openai

###############################################################################
# GITHUB ACTIONS — one-time OIDC setup (run once per subscription, no resource
# groups need to exist yet — subscription scope handles ephemeral rg-robe-*)
###############################################################################

# 1. Create the service principal scoped to the subscription
az ad sp create-for-rbac --name "sp-robe-github-deploy" \
  --role Contributor \
  --scopes /subscriptions/<SUB_ID>
# Output includes clientId, tenantId, subscriptionId — save for step 3.

# 2. Add federated credential so GitHub Actions can mint OIDC tokens for it
az ad app federated-credential create --id <CLIENT_ID> --parameters '{
  "name": "robe-github-main",
  "issuer": "https://token.actions.githubusercontent.com",
  "subject": "repo:ekaag/robe.ai:ref:refs/heads/main",
  "audiences": ["api://AzureADTokenExchange"]
}'

# 3. In GitHub → Settings → Environments, create environments: dev, gamma, live
#    Add these secrets to each (values from step 1 output):
#      AZURE_CLIENT_ID       → clientId
#      AZURE_TENANT_ID       → tenantId
#      AZURE_SUBSCRIPTION_ID → subscriptionId
#    Add these variables to gamma/live (leave empty on dev → uses FakeAuthProvider):
#      FRONTEND_ENTRA_AUTHORITY
#      FRONTEND_ENTRA_CLIENT_ID
#      FRONTEND_ENTRA_API_SCOPE

###############################################################################
# FRONTEND DEPLOY — any stage (build Next.js → push to Azure SWA)
###############################################################################
# Prereq: Bicep infra for that stage must already exist (SWA resource created).
# For dev: dev-create.sh already does this. Use deploy-frontend.sh to redeploy
# just the frontend, or to deploy gamma/live after main.bicep has run.
#
# PREFERRED: trigger via GitHub Actions (runs on Linux — no Windows EPERM issue):
gh workflow run deploy-frontend.yml --field stage=dev
gh workflow run deploy-frontend.yml --field stage=gamma
gh workflow run deploy-frontend.yml --field stage=live

# Watch the run:
gh run list --workflow=deploy-frontend.yml
gh run watch                          # picks the latest run

# Pushes to main that change vestra/** auto-deploy the dev stage — no manual trigger needed.

# Manual (Linux/WSL/CI only — EPERM on Windows without Developer Mode):

cd infra/Azure/bicep

# Redeploy dev frontend only (e.g. after a UI change):
./deploy-frontend.sh --stage=dev

# Deploy gamma frontend (Bicep infra must exist in rg-robe-gamma):
./deploy-frontend.sh --stage=gamma

# Deploy live frontend (Bicep infra must exist in rg-robe-live):
./deploy-frontend.sh --stage=live --yes

# With real Entra sign-in (needed for gamma/live — set env vars before running):
export FRONTEND_ENTRA_AUTHORITY="https://vestraoauth.ciamlogin.com/vestraoauth.onmicrosoft.com"
export FRONTEND_ENTRA_CLIENT_ID="d4cd6bf9-9a9e-44e6-b7e5-12660e5e32d9"
export FRONTEND_ENTRA_API_SCOPE="api://d4cd6bf9-9a9e-44e6-b7e5-12660e5e32d9/access_as_user"
./deploy-frontend.sh --stage=gamma --yes

###############################################################################
# GAMMA + LIVE — infra deploy (Bicep only, no frontend content)
###############################################################################
# main.bicep creates all 3 resource groups + all 3 stages at once.
# After deploy, populate secrets manually for gamma/live (dev-create.sh does
# this automatically for dev only).

az login
az account set --subscription <subscription-id>

# Deploy all 3 stages (creates rg-robe-dev, rg-robe-gamma, rg-robe-live):
az deployment sub create \
  --location canadacentral \
  --template-file infra/Azure/bicep/main.bicep

# Deploy a single stage (e.g. gamma) into its existing resource group:
az deployment group create \
  --resource-group rg-robe-gamma \
  --template-file infra/Azure/bicep/modules/stage.bicep \
  --parameters infra/Azure/bicep/parameters/gamma.bicepparam

# Populate secrets after Bicep deploy (gamma example — repeat for live):
AOAI_ENDPOINT=$(az deployment group show \
  --resource-group rg-robe-gamma \
  --name deploy-stage-gamma \
  --query "properties.outputs.openAiEndpoint.value" -o tsv)

az keyvault secret set --vault-name kv-robeai-gamma --name AzureOpenAI--Endpoint      --value "$AOAI_ENDPOINT"
az keyvault secret set --vault-name kv-robeai-gamma --name AzureOpenAI--DeploymentName --value "gpt-5-mini"
# No API key needed — App Service managed identity uses Cognitive Services OpenAI User RBAC.

# Deploy robe.api to gamma App Service:
dotnet publish robe.api -c Release -o /tmp/robe-publish
cd /tmp/robe-publish && zip -r ../robe.zip . && cd -
az webapp deploy --resource-group rg-robe-gamma --name app-robe-gamma \
  --src-path /tmp/robe.zip --type zip

# Then deploy frontend:
./infra/Azure/bicep/deploy-frontend.sh --stage=gamma --yes
