# GAMMA STAGE — full environment deploy (infra + backend + frontend)
###############################################################################

# Step 1: Deploy infra (Bicep — creates rg-robe-gamma and all its resources)
az login
az account set --subscription c3b2ab0c-a402-4b3a-aed1-c62aca631c9e
# Deploy gamma only:
az deployment group create \
  --resource-group rg-robe-gamma \
  --template-file infra/Azure/bicep/modules/stage.bicep \
  --parameters infra/Azure/bicep/parameters/gamma.bicepparam
# (Or deploy all 3 stages at once via main.bicep — see bottom of file)

# Step 2: Populate Key Vault secrets (gamma)
AOAI_ENDPOINT=$(az deployment group show \
  --resource-group rg-robe-gamma \
  --name deploy-stage-gamma \
  --query "properties.outputs.openAiEndpoint.value" -o tsv)
az keyvault secret set --vault-name kv-robeai-gamma --name AzureOpenAI--Endpoint       --value "$AOAI_ENDPOINT"
az keyvault secret set --vault-name kv-robeai-gamma --name AzureOpenAI--DeploymentName --value "gpt-5-mini"
# No API key needed — managed identity uses Cognitive Services OpenAI User RBAC.

COSMOS_ENDPOINT=$(az deployment group show \
  --resource-group rg-robe-gamma \
  --name deploy-stage-gamma \
  --query "properties.outputs.cosmosEndpoint.value" -o tsv)
STORAGE_BLOB_URI=$(az deployment group show \
  --resource-group rg-robe-gamma \
  --name deploy-stage-gamma \
  --query "properties.outputs.storageBlobServiceUri.value" -o tsv)
az keyvault secret set --vault-name kv-robeai-gamma --name CosmosDb--Endpoint       --value "$COSMOS_ENDPOINT"
az keyvault secret set --vault-name kv-robeai-gamma --name CosmosDb--DatabaseName   --value "robeai"
az keyvault secret set --vault-name kv-robeai-gamma --name CosmosDb--ContainerName  --value "garments"
az keyvault secret set --vault-name kv-robeai-gamma --name Storage--BlobServiceUri  --value "$STORAGE_BLOB_URI"
az keyvault secret set --vault-name kv-robeai-gamma --name Storage--ContainerName   --value "garment-images"
# No keys/connection strings needed — managed identity uses Cosmos DB Built-in Data
# Contributor + Storage Blob Data Contributor RBAC.

# Step 3: Deploy robe.api to gamma App Service
dotnet publish robe.api -c Release -o /tmp/robe-publish
cd /tmp/robe-publish && zip -r ../robe.zip . && cd -
az webapp deploy --resource-group rg-robe-gamma --name app-robe-gamma \
  --src-path /tmp/robe.zip --type zip

# Step 4: Deploy frontend via GitHub Actions
# Prereq: set Entra variables in GitHub → Settings → Environments → gamma:
#   FRONTEND_ENTRA_AUTHORITY, FRONTEND_ENTRA_CLIENT_ID, FRONTEND_ENTRA_API_SCOPE
gh workflow run deploy-frontend.yml --field stage=gamma
gh run watch
